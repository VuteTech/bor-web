{{- /* Fingerprinted screenshot image with WebP variants. Usage: {{ partial "screenshot.html" (dict "file" "dashboard.png" "alt" "Bor Dashboard") }} The gallery shows images at the container width (up to 1344px), so the srcset offers 960, 1600 and the native width. */ -}} {{- $img := resources.Get (printf "images/screenshots/%s" .file) -}} {{- if not $img -}}{{ errorf "screenshot.html: images/screenshots/%s not found in assets" .file }}{{- end -}} {{- $set := slice -}} {{- range slice 960 1600 -}} {{- if lt . $img.Width -}} {{- $v := $img.Resize (printf "%dx webp q82" .) -}} {{- $set = $set | append (printf "%s %dw" $v.RelPermalink .) -}} {{- end -}} {{- end -}} {{- $native := $img.Resize (printf "%dx webp q82" $img.Width) -}} {{- $set = $set | append (printf "%s %dw" $native.RelPermalink $img.Width) -}}
